name: Build Android APK

# Runs by itself whenever you upload changes, and you can also start it by hand
# from the "Actions" tab (choose "Build Android APK", then "Run workflow").
on:
  push:
    branches: [main, master]
  workflow_dispatch:

permissions:
  contents: write # lets the build attach the APK to a Release

jobs:
  apk:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm

      # The Android build tools need Java 21.
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '21'

      - name: Install and build the web app
        run: |
          npm ci
          npm run build

      - name: Create the Android project
        run: npx cap add android

      # App icon and start-up screen. If this step ever fails, the APK still builds
      # with the default icon.
      - name: Make app icons
        continue-on-error: true
        run: npx capacitor-assets generate --android

      - name: Copy the web app into the Android project
        run: npx cap sync android

      # Every build gets a higher version number so a new APK installs over the old one.
      - name: Set the version number
        run: sed -i "s/versionCode 1/versionCode ${{ github.run_number }}/; s/versionName \"1.0\"/versionName \"1.0.${{ github.run_number }}\"/" android/app/build.gradle

      # Signing with the same key every time is what lets updates install on top of the
      # old app WITHOUT erasing your data.
      - name: Use the fixed signing key
        run: |
          mkdir -p ~/.android
          cp signing/debug.keystore ~/.android/debug.keystore

      - name: Make sure the Android SDK parts are installed
        continue-on-error: true
        run: |
          yes | "$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager" --licenses > /dev/null || true
          "$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager" "platforms;android-36" "build-tools;36.0.0" || true

      - name: Build the APK
        working-directory: android
        run: ./gradlew assembleDebug --no-daemon

      - name: Name the file
        run: cp android/app/build/outputs/apk/debug/app-debug.apk Daymark.apk

      # Easiest download: open the "Releases" section of the repository.
      - name: Publish to Releases
        continue-on-error: true
        uses: softprops/action-gh-release@v2
        with:
          tag_name: apk-${{ github.run_number }}
          name: Daymark APK (build ${{ github.run_number }})
          files: Daymark.apk

      # Backup download: open the finished run, "Artifacts" at the bottom.
      - uses: actions/upload-artifact@v4
        with:
          name: Daymark-APK
          path: Daymark.apk
